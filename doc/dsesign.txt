// ─── Border helpers ───────────────────────────────────────────────────────────
const border = (color = C.border, size = 1) => ({ style: BorderStyle.SINGLE, size, color });
const noBorder = () => ({ style: BorderStyle.NONE, size: 0, color: "FFFFFF" });

const allBorders = (color, size) => ({
  top: border(color, size), bottom: border(color, size),
  left: border(color, size), right: border(color, size)
});

const noBorders = () => ({
  top: noBorder(), bottom: noBorder(), left: noBorder(), right: noBorder()
});

// ─── Spacing helpers ──────────────────────────────────────────────────────────
const sp = (before = 0, after = 0) => ({ before, after });

// ─── Text helpers ─────────────────────────────────────────────────────────────
const run = (text, opts = {}) => new TextRun({ text, font: "Arial", ...opts });
const bold = (text, opts = {}) => run(text, { bold: true, ...opts });
const mono = (text, opts = {}) => new TextRun({ text, font: "Courier New", size: 18, ...opts });
const gold = (text, opts = {}) => bold(text, { color: C.gold, ...opts });
const navy = (text, opts = {}) => bold(text, { color: C.navy, ...opts });
const muted = (text, opts = {}) => run(text, { color: C.muted, ...opts });
const link = (text, url) => new ExternalHyperlink({ link: url, children: [new TextRun({ text, font: "Arial", color: C.accent, underline: {} })] });

// ─── Paragraph helpers ────────────────────────────────────────────────────────
const h1 = (text) => new Paragraph({

const pStr = (text, opts = {}) => new Paragraph({ spacing: sp(80, 160), children: [run(text, opts)] });

const bullet = (text, opts = {}) => new Paragraph({
  numbering: { reference: "bullets", level: 0 },
  spacing: sp(60, 60),
  children: [run(text, { size: 22, ...opts })]
});

const sub_bullet = (text, opts = {}) => new Paragraph({
  numbering: { reference: "subbullets", level: 1 },
  spacing: sp(40, 40),
  children: [run(text, { size: 20, color: C.muted, ...opts })]
});

const numbered = (text, opts = {}) => new Paragraph({
  numbering: { reference: "numbers", level: 0 },
  spacing: sp(60, 60),
  children: [run(text, { size: 22, ...opts })]
});

const spacer = (n = 1) => Array.from({ length: n }, () => new Paragraph({ children: [run("")], spacing: sp(0, 0) }));

const divider = () => new Paragraph({
  spacing: sp(200, 200),
  border: { bottom: { style: BorderStyle.SINGLE, size: 2, color: C.border } },
  children: [run("")]
});

// ─── Code block ───────────────────────────────────────────────────────────────
const codeBlock = (lines) => {
  return lines.map((line, i) => new Paragraph({
    spacing: sp(i === 0 ? 80 : 0, i === lines.length - 1 ? 80 : 0),
    shading: { fill: C.codeGray, type: ShadingType.CLEAR },
    border: i === 0 ? {
      top: border(C.border, 1), left: border(C.gold, 4), right: border(C.border, 1),
      bottom: lines.length === 1 ? border(C.border, 1) : noBorder()
    } : i === lines.length - 1 ? {
      top: noBorder(), left: border(C.gold, 4), right: border(C.border, 1),
      bottom: border(C.border, 1)
    } : {
      top: noBorder(), bottom: noBorder(),
      left: border(C.gold, 4), right: border(C.border, 1)
    },
    children: [new TextRun({ text: line || " ", font: "Courier New", size: 18, color: C.slate })]
  }));
};

// ─── Section callout (tip / warning / critical) ───────────────────────────────
const callout = (type, text) => {
  const map = {
    tip:      { fill: "F0FDF4", accent: "166534", label: "TIP",      icon: "💡" },
    warning:  { fill: "FFFBEB", accent: "92400E", label: "WARNING",  icon: "⚠️" },
    critical: { fill: "FEF2F2", accent: "991B1B", label: "CRITICAL", icon: "🚨" },
    info:     { fill: "EFF6FF", accent: "1E40AF", label: "NOTE",     icon: "ℹ️" },
  };
  const { fill, accent, label, icon } = map[type] || map.info;
  return new Table({
    width: { size: 9360, type: WidthType.DXA },
    columnWidths: [9360],
    rows: [new TableRow({
      children: [new TableCell({
        borders: { ...noBorders(), left: { style: BorderStyle.SINGLE, size: 8, color: accent } },
        shading: { fill, type: ShadingType.CLEAR },
        margins: { top: 100, bottom: 100, left: 200, right: 160 },
        children: [new Paragraph({ children: [bold(`${icon}  ${label}: `, { size: 20, color: accent }), run(text, { size: 20, color: accent })] })]
      })]
    })]
  });
};
